<!-- typst.ts singleton pattern from RESEARCH.md Pattern 4 (exact API): -->
```typescript
import { $typst } from '@myriaddreamin/typst.ts/dist/esm/contrib/snippet.mjs'
import compilerWasm from '@myriaddreamin/typst-ts-web-compiler/pkg/typst_ts_web_compiler_bg.wasm?url'
import rendererWasm from '@myriaddreamin/typst-ts-renderer/pkg/typst_ts_renderer_bg.wasm?url'

let initPromise: Promise<void> | null = null

async function ensureTypstReady(): Promise<void> {
  if (initPromise !== null) return initPromise
  initPromise = (async () => {
    $typst.setCompilerInitOptions({ getModule: () => compilerWasm })
    $typst.setRendererInitOptions({ getModule: () => rendererWasm })
  })()
  return initPromise
}

export async function renderTypstToSvg(source: string): Promise<string> {
  await ensureTypstReady()
  return $typst.svg({ mainContent: source })
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| card content → #typst parser | The raw Typst source from the card content flows into the WASM compiler and the SVG output is injected via dangerouslySetInnerHTML |
| typst.ts WASM → DOM | SVG output from $typst.svg() is injected directly into the DOM; typst.ts is assumed to sanitize (per RESEARCH.md A2 assumption) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-SVG | Spoofing / Tampering | dangerouslySetInnerHTML in TypstBlock | accept | XSS via malicious Typst SVG output — mitigated by invite-only user model (2-5 trusted users). Full SVG sanitization is a Phase 5+ concern. Document assumption A2 from RESEARCH.md |
| T-03-WASM | Denial of Service | Typst WASM lazy init | mitigate | initPromise singleton ensures WASM loads only once. Failed init should be caught and reflected as an error state rather than crashing the app |
| T-03-03 | Tampering | #typst detection (h6 vs p) | mitigate | Handle both h6 and p component paths to prevent Typst source leaking as visible headings if Markdown parsing behavior differs |
| T-03-SC | Tampering | npm/pip/cargo installs | mitigate | @myriaddreamin packages verified [OK] in RESEARCH.md Package Legitimacy Audit. RC status (0.7.0-rc2) noted — pin exact version in package.json |
</threat_model>
