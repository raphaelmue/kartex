```typescript
import ReactMarkdown from 'react-markdown'
import remarkGfm from 'remark-gfm'

interface KartexRendererProps {
  content: string
}

/**
 * KartexRenderer — renders Kartex card content as HTML.
 *
 * Phase 2: Markdown only (react-markdown + remark-gfm).
 * Phase 3 extension point: add `components` prop to ReactMarkdown for KaTeX math
 * nodes and Typst WASM blocks — the external interface (content: string) must not change.
 */
export function KartexRenderer({ content }: KartexRendererProps) {
  return (
    <div className="prose prose-sm max-w-none dark:prose-invert">
      <ReactMarkdown remarkPlugins={[remarkGfm]}>
        {content}
      </ReactMarkdown>
    </div>
  )
}
```

**Critical constraints:**
- Use `remarkPlugins={[remarkGfm]}` (NOT the deprecated `plugins` prop from react-markdown v7/v8)
- Use `components` prop (NOT `renderers`) for any future customization — do NOT use `renderers` here
- The `className="prose prose-sm max-w-none dark:prose-invert"` wrapper is required per UI-SPEC.md (Tailwind Typography prose classes)
- The component is a named export (not default export) — consumers use `import { KartexRenderer } from '@/components/KartexRenderer'`
- Do NOT add any KaTeX or Typst logic in Phase 2 — this component is intentionally Markdown-only now
