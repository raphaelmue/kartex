    const sw = readFileSync('./public/sw.js', 'utf8')
    c.header('Cache-Control', 'no-store')
    c.header('Content-Type', 'application/javascript')
    return c.body(sw)
  } catch {
    return c.text('Service worker not found', 404)

})

    onFound: (path, c) => {

        c.header('Cache-Control', 'no-store')
      }
    },
  }),
)
