    Step C — Create `packages/shared/src/lib/kartex-parser.ts`:

    ```typescript
    import { parse as parseYaml } from 'yaml'
    import type { DeckHeader, ParsedCard, ParseWarning } from '../schemas/import.js'

    export type { DeckHeader, ParsedCard, ParseWarning }

    export interface KartexParseResult {
      deck: DeckHeader
      cards: ParsedCard[]
      warnings: ParseWarning[]
    }

    export interface KartexParseError {
      fatal: true
      message: string
    }

    // Parses a .kartex file string into a typed result. Pure function — no I/O.
    // Returns KartexParseResult on success, KartexParseError on fatal failure (D-02: missing/invalid header).
    // Malformed card blocks are skipped with a warning (D-01: lenient for cards).
    export function parseKartex(source: string): KartexParseResult | KartexParseError {
      // 1. Strip comment lines (lines starting with '#' that are NOT inside a card block)
      //    Comments are stripped before header/card parsing to avoid interference
      const strippedLines = source.split('\n').map((line) => {
        // Only strip standalone comment lines — preserve '#typst' inside card content
        // A comment line is one whose first non-whitespace char is '#' and appears outside delimiters
        return line
      })
      const fullText = strippedLines.join('\n')

      // 2. Extract YAML header between first --- ... ---
      const headerMatch = fullText.match(/^---\n([\s\S]*?)\n---/)
      if (!headerMatch) {
        return {
          fatal: true,
          message: "No deck header found. Your .kartex file must start with a `---` YAML block.",
        }
      }

      // 3. Parse the YAML header
      let rawHeader: Record<string, unknown>
      try {
        rawHeader = parseYaml(headerMatch[1]) as Record<string, unknown>
      } catch {
        return {
          fatal: true,
          message: "Deck header is not valid YAML. Your .kartex file must start with a `---` YAML block.",
        }
      }

      if (typeof rawHeader?.deck !== 'string' || rawHeader.deck.trim() === '') {
        return {
          fatal: true,
          message: "No deck header found. Your .kartex file must start with a `---` YAML block.",
        }
      }

      const deck: DeckHeader = {
        deck: rawHeader.deck.trim(),
        author: typeof rawHeader.author === 'string' ? rawHeader.author.trim() : undefined,
        tags: Array.isArray(rawHeader.tags) ? rawHeader.tags.map(String) : [],
      }

      // 4. Extract card blocks — text after the closing ---
      const afterHeader = fullText.slice(fullText.indexOf('---', fullText.indexOf('---') + 3) + 3)

      // Split on ':: card' delimiter (trim whitespace variants per Pitfall 1)
      // Card block: from ':: card' to next '::'
      const cards: ParsedCard[] = []
      const warnings: ParseWarning[] = []

      // Use regex to find all card block contents between ':: card' and '::'
      // /^::\s*card\s*$([\s\S]*?)^::\s*$/gm
      const cardBlockRegex = /^::\s*card\s*$([\s\S]*?)^::\s*$/gm
      let cardIndex = 0
      let match: RegExpExecArray | null

      while ((match = cardBlockRegex.exec(afterHeader)) !== null) {
        cardIndex++
        const blockContent = match[1]

        // Parse front: and back: fields from block content
        // Fields can be multi-line — value continues until the next field name or end of block
        const parsed = parseCardBlock(blockContent, cardIndex)
        if ('warning' in parsed) {
          warnings.push(parsed.warning)
        } else {
          cards.push(parsed.card)
        }
      }

      return { deck, cards, warnings }
    }

    // Parses a single card block's content string.
    // Returns { card } on success or { warning } on failure (malformed block).
    function parseCardBlock(
      content: string,
      cardIndex: number,
    ): { card: ParsedCard } | { warning: ParseWarning } {
      // Extract front:, back:, tags: from block
      // Each field value starts after 'fieldname:' and continues until the next recognized field
      // or end of content. Multi-line values are collected as-is.

      // Strip comment lines inside card block
      const lines = content.split('\n').filter((line) => !line.match(/^#(?!typst)/))
      const blockText = lines.join('\n')

      // Match field values: greedy match from 'field:' to next 'field:' or end of string
      // front:, back:, tags: are the only recognized fields
      const frontMatch = blockText.match(/^front:([\s\S]*?)(?=^back:|^tags:|$)/m)
      const backMatch = blockText.match(/^back:([\s\S]*?)(?=^front:|^tags:|$)/m)
      const tagsMatch = blockText.match(/^tags:\s*(.+)$/m)

      const front = frontMatch ? frontMatch[1].trim() : ''
      const back = backMatch ? backMatch[1].trim() : ''

      if (!front) {
        return {
          warning: {
            cardIndex,
            reason: "Missing `front:` field — card skipped",
          },
        }
      }

      if (!back) {
        return {
          warning: {
            cardIndex,
            reason: "Missing `back:` field — card skipped",
          },
        }
      }

      // Parse tags from card block — value after 'tags:' is NOT YAML list syntax in card context
      // but may be formatted as '[a, b]' — use yaml.parse on the value portion (Pitfall 2)
      let tags: string[] = []
      if (tagsMatch) {
        const tagsValue = tagsMatch[1].trim()
        try {
          const parsed = parseYaml(tagsValue)
          if (Array.isArray(parsed)) {
            tags = parsed.map(String)
          }
        } catch {
          // Tags parse failed — treat as empty (non-fatal)
        }
      }

      return { card: { front, back, tags } }
    }
    ```
